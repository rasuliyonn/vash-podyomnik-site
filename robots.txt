# robots.txt — «Ваш Подъёмник»
# Служебный каталог /data закрыт. Админку НЕ закрываем Disallow: иначе поисковик
# не увидит meta/X-Robots-Tag noindex на её страницах и URL может попасть в индекс
# без описания. Админка закрыта noindex (meta + заголовок .htaccess) — этого достаточно.
# Ниже — явные правила для ответных ИИ-краулеров (GEO). Content-Signal задаёт
# предпочтение: поиск и ответы (ai-input) — да, обучение моделей (ai-train) — нет.
# Content-Signal — это сигнал (Cloudflare Content Signals Policy), не
# принудительное ограничение; Google его не учитывает.

User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=no
Allow: /
Disallow: /data
